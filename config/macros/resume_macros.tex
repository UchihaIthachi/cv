\newcommand{\ProjectTitle}[1]{%

}

\newcommand{\cvproject}[4]{
    \ifProjectNumbering
        \stepcounter{projectCount}%
        \textbf{\theprojectCount. #1} \hfill #2 \par
    \else
        \textbf{#1} \hfill #2 \par
    \fi
    \vspace{-0.5em}
    \begin{itemize}
        \item #3
        \item \textbf{Technologies:} #4
    \end{itemize}
    \vspace{-0.5em}
}
