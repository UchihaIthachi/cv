% Project Numbering Logic
\newif\ifProjectNumbering
\ProjectNumberingfalse % Default to false
\newcounter{projectCount}

\newcommand{\ProjectTitle}[1]{%
  \ifProjectNumbering
    \stepcounter{projectCount}%
    \subsection*{\theprojectCount. #1}%
  \else
    \subsection*{#1}%
  \fi
}

\newcommand{\cvproject}[4]{
    \ifProjectNumbering
        \stepcounter{projectCount}%
        \textbf{\theprojectCount. #1} \hfill \textit{#2} \par
    \else
        \textbf{#1} \hfill \textit{#2} \par
    \fi
    
    
    #3 \par
    
    
    \textbf{Technologies:} #4 \par
    \vspace{0.5em}
    \vspace{0.5em}
}

\definecolor{BlockGray}{HTML}{D9D9D9}

\newcommand{\titleblock}[1]{%
  \begingroup
  \setlength{\fboxsep}{1pt}%
  \textbf{#1}%
  \endgroup
}

\newcommand{\cvprojectfull}[4]{%
  \noindent\titleblock{#1}%
  \if\relax\detokenize{#2}\relax
  \else
    \hfill{\scriptsize\textit{#2}}%
  \fi
  \par
  \vspace{0.08em}
  \begin{itemize}[
    leftmargin=*,
    itemsep=0pt,
    topsep=0pt,
    parsep=0pt,
    partopsep=0pt
  ]
    \item #3
    \item #4
  \end{itemize}
  \vspace{-0.35em}
}
