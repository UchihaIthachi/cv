% ATS-safe keyword styles
% Simplified \skw to bold-only (dropped underline). Underlined text in a
% resume commonly reads as "this is a hyperlink" to both humans and some
% parsers, which adds visual noise without adding ATS keyword weight --
% bold alone is picked up identically by keyword-matching engines.

\newcommand{\skw}[1]{\textbf{#1}}

    \else
        \textbf{#1} \hfill #2 \par
    \fi

        \item #3
        \item \textbf{Technologies:} #4
    \end{itemize}
    \vspace{-0.5em}
}

% CHANGED: projects now stack full-width, single column, in normal
% top-to-bottom reading order (was: two-column tabularx grid).
% Why: PDF text-extraction (what ATS parsers actually read) walks a
% page roughly left-to-right in horizontal bands. In a 2-column table,
% that often interleaves Project A's text with Project B's text mid-
% sentence, corrupting keyword context (e.g. "Technologies" line from
% one project gets glued to bullet text from the other). A single
% vertical stack reads in the exact order it's written, every time --
% all 6 projects are kept, nothing removed, just re-flowed.
\newcommand{\cvprojectfull}[4]{%
  \noindent \titleblock{#1} \hfill {\scriptsize\textit{#2}} \\[0.15em]
  \begin{itemize}[leftmargin=*, itemsep=0pt, topsep=1pt, parsep=0pt]
    \item #3
    \item #4
  \end{itemize}
  \vspace{0.28em}
}
